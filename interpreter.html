// THE REVERSE SEARCH ENGINE function searchStrongs() { let query = document.getElementById('strongsInput').value.trim(); if(!query) return; apiResults.innerHTML = ''; // Failsafe: Ensure dictionary is loaded if (typeof strongsGreekDictionary === 'undefined') { apiResults.innerHTML = `
Database not loaded. Check file connection.
`; return; } let upperQuery = query.toUpperCase(); // 1. Direct Strong's Number Search (e.g., "G2424" or just "2424") let strongKey = upperQuery.startsWith('G') ? upperQuery : 'G' + upperQuery; if (strongsGreekDictionary[strongKey] && !isNaN(upperQuery.replace('G', ''))) { renderGreekCard(strongKey); return; } // 2. English Keyword Reverse Search let results = []; let searchWord = query.toLowerCase(); // Loop through the entire 5,000+ word dictionary instantly for (const [key, data] of Object.entries(strongsGreekDictionary)) { let kjv = (data.kjv_def || "").toLowerCase(); let def = (data.strongs_def || "").toLowerCase(); // Use regex to find whole words only (so "son" doesn't match "person") let regex = new RegExp(`\\b${searchWord}\\b`, 'i'); if (regex.test(kjv) || regex.test(def)) { results.push({ key: key, data: data }); } } // 3. Render Search Results if (results.length === 0) { apiResults.innerHTML = `
No Greek matches found for "${query}".
`; return; } // If exactly one match, go straight to the card if (results.length === 1) { renderGreekCard(results[0].key); return; } // If multiple matches, generate a clickable list let html = `

Found ${results.length} matches for "${query}"

`; results.forEach(res => { let preview = res.data.kjv_def ? res.data.kjv_def.substring(0, 50) + '...' : 'Click to view details'; html += `
${res.data.lemma} ${res.key}
${res.data.translit || ''}
${preview}
`; }); apiResults.innerHTML = html; } // THE LEXICON CARD RENDERER function renderGreekCard(key) { const data = strongsGreekDictionary[key]; const defText = data.strongs_def || "Definition not provided in database."; const derivationText = data.derivation ? `
Derivation: ${data.derivation}
` : ""; const kjvText = data.kjv_def ? `
KJV TRANSLATION(S)
${data.kjv_def}
` : ""; apiResults.innerHTML = `

${data.lemma} ${key}

${data.translit || ''}
${derivationText}
STRONG'S DEFINITION
${defText}
${kjvText} `; }
